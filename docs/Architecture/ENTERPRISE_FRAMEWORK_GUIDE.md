# Enterprise Framework Developer Guide

🚀 **Complete guide for developers working with the Enterprise Framework generated by Enterprise CLI.**

The Enterprise Framework is a revolutionary approach that **forces developers to write only business logic** while making all infrastructure code impossible to modify. This guide explains how to work effectively within this constrained but powerful environment.

## 🎯 **Core Philosophy**

### Traditional Development vs Enterprise Framework

**❌ Traditional Development (Before):**
```java
@Service
@Transactional  // Can forget
public class CustomerService {

    // Can implement CRUD incorrectly
    public Customer create(CustomerRequest request) {
        // No validation enforcement
        // Manual caching management
        // Manual metrics collection
        // Manual audit logging
        // Manual event publishing
        // Mixed business + infrastructure concerns
    }
}
```

**✅ Enterprise Framework (After):**
```java
@Service
public class CustomerServiceImpl extends AbstractEnterpriseService<Customer, String, CustomerRequest, CustomerRequest, CustomerResponse> {

    // 🤖 INFRASTRUCTURE: Automatic (cannot be modified)
    // - Transactions ✓  - Caching ✓  - Metrics ✓  - Audit ✓  - Events ✓

    // 👨‍💻 BUSINESS LOGIC: Required implementation (pure business logic only)
    @Override
    protected Customer createEntityFromRequest(CustomerRequest createDTO) {
        return customerMapper.toEntity(createDTO); // Pure business mapping
    }

    @Override
    protected CustomerRepository getRepository() {
        return customerRepository; // Dependency injection
    }
}
```

### 2. Business Validation Implementation

```java
@Component
public class CustomerBusinessValidator
    extends BusinessValidator.AbstractBusinessValidator<Customer, CustomerRequest, CustomerRequest> {

    private final CustomerRepository customerRepository;

    @Autowired
    public CustomerBusinessValidator(CustomerRepository customerRepository) {
        this.customerRepository = customerRepository;
    }

    @Override
    public ValidationResult validateCreate(CustomerRequest createDTO) {
        return validationBuilder()
                // 👨‍💻 Basic field validations
                .checkNotEmpty(createDTO.getEmail(), "Customer email")
                .checkNotEmpty(createDTO.getName(), "Customer name")
                .checkMinLength(createDTO.getName(), 2, "Customer name")
                .checkMaxLength(createDTO.getName(), 100, "Customer name")

                // 👨‍💻 Business rule validations
                .check(isValidEmail(createDTO.getEmail()), "Invalid email format")
                .check(!customerRepository.existsByEmail(createDTO.getEmail()),
                       "Customer with email '" + createDTO.getEmail() + "' already exists")

                // 👨‍💻 Business-specific validations
                .check(isValidBusinessEmail(createDTO.getEmail()),
                       "Email must be from approved business domains")

                // 👨‍💻 Warnings (non-blocking)
                .warn(createDTO.getPhone() == null || createDTO.getPhone().trim().isEmpty(),
                      "Phone number is recommended for better customer communication")

                .build();
    }

    @Override
    public ValidationResult validateUpdate(Object entityId, CustomerRequest updateDTO, Customer existingEntity) {
        return validationBuilder()
                // 👨‍💻 Same field validations as create
                .checkNotEmpty(updateDTO.getEmail(), "Customer email")
                .checkNotEmpty(updateDTO.getName(), "Customer name")

                // 👨‍💻 Update-specific validations
                .check(!isEmailChangedToExisting(updateDTO.getEmail(), existingEntity),
                       "Email '" + updateDTO.getEmail() + "' is already used by another customer")

                // 👨‍💻 Business state validations
                .check(canUpdateCustomer(existingEntity),
                       "Customer cannot be updated in current state: " + existingEntity.getStatus())

                .build();
    }

    @Override
    public ValidationResult validateDelete(Object entityId, Customer entity) {
        return validationBuilder()
                // 👨‍💻 Deletion business rules
                .check(!hasActiveOrders(entity),
                       "Cannot delete customer with active orders")

                .check(!"VIP".equals(entity.getMembershipLevel()),
                       "VIP customers cannot be deleted without approval")

                .build();
    }

    // 👨‍💻 Private business validation methods
    private boolean isValidBusinessEmail(String email) {
        String[] approvedDomains = {"company.com", "enterprise.com", "business.org"};
        String domain = email.substring(email.indexOf("@") + 1);
        return Arrays.asList(approvedDomains).contains(domain);
    }

    private boolean isEmailChangedToExisting(String newEmail, Customer existingEntity) {
        if (existingEntity.getEmail().equalsIgnoreCase(newEmail)) {
            return false; // Same email, no change
        }
        return customerRepository.existsByEmail(newEmail);
    }

    private boolean canUpdateCustomer(Customer customer) {
        return !"LOCKED".equals(customer.getStatus()) && !"ARCHIVED".equals(customer.getStatus());
    }

    private boolean hasActiveOrders(Customer customer) {
        // Implement check for active orders
        return false; // Simplified for example
    }
}
```

### 3. Business Rules Implementation

```java
@Component
public class CustomerBusinessRules
    extends BusinessRules.AbstractBusinessRules<Customer, CustomerRequest, CustomerRequest> {

    private static final Logger logger = LoggerFactory.getLogger(CustomerBusinessRules.class);

    @Override
    public void applyCreateRules(Customer entity, CustomerRequest createDTO) {
        logger.debug("🔧 Applying create rules for customer: {}", createDTO.getEmail());

        ruleBuilder()
                // 👨‍💻 Rule: Generate business identifier
                .rule("Generate Business ID",
                      () -> entity.getBusinessId() == null,
                      () -> entity.setBusinessId(generateBusinessId("CUST", entity)))

                // 👨‍💻 Rule: Set default status
                .rule("Set Initial Status",
                      () -> entity.getStatus() == null,
                      () -> entity.setStatus("PENDING_VERIFICATION"))

                // 👨‍💻 Rule: Calculate membership level
                .rule("Calculate Membership Level",
                      () -> entity.getMembershipLevel() == null,
                      () -> entity.setMembershipLevel(calculateInitialMembershipLevel(createDTO)))

                // 👨‍💻 Rule: Set business defaults
                .rule("Apply Business Defaults",
                      () -> needsBusinessDefaults(createDTO),
                      () -> applyBusinessDefaults(entity, createDTO))

                .executeWithLogging(logger);

        logger.debug("✅ Create rules applied successfully for customer: {}", entity.getBusinessId());
    }

    @Override
    public void applyUpdateRules(Customer entity, CustomerRequest updateDTO, Customer previousState) {
        logger.debug("🔧 Applying update rules for customer: {}", entity.getBusinessId());

        ruleBuilder()
                // 👨‍💻 Rule: Handle email changes
                .rule("Process Email Change",
                      () -> hasChanged(previousState.getEmail(), entity.getEmail()),
                      () -> handleEmailChange(entity, previousState))

                // 👨‍💻 Rule: Update membership level
                .rule("Recalculate Membership Level",
                      () -> shouldRecalculateMembership(entity, updateDTO),
                      () -> entity.setMembershipLevel(calculateMembershipLevel(entity, updateDTO)))

                // 👨‍💻 Rule: Handle status transitions
                .rule("Process Status Change",
                      () -> hasChanged(previousState.getStatus(), entity.getStatus()),
                      () -> processStatusTransition(entity, previousState.getStatus(), entity.getStatus()))

                .executeWithLogging(logger);

        logger.debug("✅ Update rules applied successfully for customer: {}", entity.getBusinessId());
    }

    @Override
    public boolean canDelete(Customer entity) {
        logger.debug("🔍 Checking if customer can be deleted: {}", entity.getBusinessId());

        boolean canDelete = true;

        // 👨‍💻 Business deletion rules
        if ("VIP".equals(entity.getMembershipLevel())) {
            logger.warn("❌ Cannot delete VIP customer: {}", entity.getBusinessId());
            canDelete = false;
        }

        if ("LOCKED".equals(entity.getStatus())) {
            logger.warn("❌ Cannot delete locked customer: {}", entity.getBusinessId());
            canDelete = false;
        }

        return canDelete;
    }

    @Override
    public void applyDeleteRules(Customer entity) {
        logger.debug("🔧 Applying delete rules for customer: {}", entity.getBusinessId());

        ruleBuilder()
                // 👨‍💻 Rule: Archive customer data
                .rule("Archive Customer Data",
                      () -> hasImportantData(entity),
                      () -> archiveCustomerData(entity))

                // 👨‍💻 Rule: Notify related systems
                .rule("Notify Integration Systems",
                      () -> hasExternalIntegrations(entity),
                      () -> notifyExternalSystems(entity))

                .executeWithLogging(logger);

        logger.debug("✅ Delete rules applied successfully for customer: {}", entity.getBusinessId());
    }

    @Override
    public Map<String, Object> calculateBusinessMetrics(Customer entity) {
        // 👨‍💻 Calculate customer-specific business metrics
        return Map.of(
                "customer_value_score", calculateCustomerValueScore(entity),
                "loyalty_index", calculateLoyaltyIndex(entity),
                "risk_factor", calculateRiskFactor(entity),
                "engagement_level", calculateEngagementLevel(entity)
        );
    }

    @Override
    public List<String> getBusinessEventsToTrigger(Customer entity, String operation) {
        List<String> events = new ArrayList<>();

        switch (operation.toUpperCase()) {
            case "CREATE" -> {
                events.add("CustomerCreated");
                events.add("NewCustomerOnboarding");
                if ("VIP".equals(entity.getMembershipLevel())) {
                    events.add("VIPCustomerCreated");
                }
            }
            case "UPDATE" -> {
                events.add("CustomerUpdated");
                if (isStatusChange(entity)) {
                    events.add("CustomerStatusChanged");
                }
            }
            case "DELETE" -> {
                events.add("CustomerDeleted");
                events.add("CustomerDataArchival");
            }
        }

        return events;
    }

    // 👨‍💻 Private business methods implementation
    private String calculateInitialMembershipLevel(CustomerRequest createDTO) {
        // Business logic for initial membership calculation
        return "STANDARD"; // Default membership level
    }

    private boolean needsBusinessDefaults(CustomerRequest createDTO) {
        return createDTO.getPreferences() == null || createDTO.getTimezone() == null;
    }

    private void applyBusinessDefaults(Customer entity, CustomerRequest createDTO) {
        if (entity.getTimezone() == null) {
            entity.setTimezone("UTC");
        }
        if (entity.getPreferences() == null) {
            entity.setPreferences("email_notifications:true,sms_notifications:false");
        }
    }

    private void handleEmailChange(Customer entity, Customer previousState) {
        // Business logic for email change
        entity.setEmailVerified(false);
        entity.setEmailChangedAt(getCurrentDateTime());
        logger.info("📧 Email changed for customer {}: {} -> {}",
                   entity.getBusinessId(), previousState.getEmail(), entity.getEmail());
    }

    private boolean shouldRecalculateMembership(Customer entity, CustomerRequest updateDTO) {
        // Determine if membership level should be recalculated
        return updateDTO.getAnnualSpending() != null || updateDTO.getOrderCount() != null;
    }

    private String calculateMembershipLevel(Customer entity, CustomerRequest updateDTO) {
        // Advanced business logic for membership calculation
        double annualSpending = updateDTO.getAnnualSpending() != null ? updateDTO.getAnnualSpending() : 0.0;
        int orderCount = updateDTO.getOrderCount() != null ? updateDTO.getOrderCount() : 0;

        if (annualSpending > 10000 || orderCount > 50) {
            return "VIP";
        } else if (annualSpending > 5000 || orderCount > 20) {
            return "PREMIUM";
        } else {
            return "STANDARD";
        }
    }

    private void processStatusTransition(Customer entity, String fromStatus, String toStatus) {
        // Business logic for status transitions
        entity.setStatusChangedAt(getCurrentDateTime());
        entity.setStatusChangeReason("Business rule triggered transition");

        // Apply status-specific business logic
        switch (toStatus) {
            case "ACTIVE" -> activateCustomer(entity);
            case "SUSPENDED" -> suspendCustomer(entity);
            case "ARCHIVED" -> archiveCustomer(entity);
        }
    }

    private void activateCustomer(Customer entity) {
        entity.setActivatedAt(getCurrentDateTime());
        logger.info("🟢 Customer activated: {}", entity.getBusinessId());
    }

    private void suspendCustomer(Customer entity) {
        entity.setSuspendedAt(getCurrentDateTime());
        logger.info("🔴 Customer suspended: {}", entity.getBusinessId());
    }

    private void archiveCustomer(Customer entity) {
        entity.setArchivedAt(getCurrentDateTime());
        logger.info("📦 Customer archived: {}", entity.getBusinessId());
    }

    // Business metrics calculation methods
    private double calculateCustomerValueScore(Customer entity) {
        // Implement customer value scoring algorithm
        return 75.0; // Simplified for example
    }

    private double calculateLoyaltyIndex(Customer entity) {
        // Implement loyalty calculation
        return 0.85; // Simplified for example
    }

    private double calculateRiskFactor(Customer entity) {
        // Implement risk assessment
        return 0.15; // Simplified for example
    }

        // Implement engagement calculation
        return "HIGH"; // Simplified for example
    }

        return entity.getOrderCount() > 0 || "VIP".equals(entity.getMembershipLevel());
```

### 4. Service Implementation

```java
@Service
public class CustomerServiceImpl
    extends AbstractEnterpriseService<Customer, String, CustomerRequest, CustomerRequest, CustomerResponse>
    implements CustomerService {

    private final CustomerRepository customerRepository;
    private final CustomerMapper customerMapper;
    private final CustomerBusinessValidator businessValidator;
    private final CustomerBusinessRules businessRules;

    @Autowired
    public CustomerServiceImpl(CustomerRepository customerRepository,
                              CustomerMapper customerMapper,
                              CustomerBusinessValidator businessValidator,
                              CustomerBusinessRules businessRules) {
        this.customerRepository = customerRepository;
        this.customerMapper = customerMapper;
        this.businessValidator = businessValidator;
        this.businessRules = businessRules;
    }

    @PostConstruct
    public void initializeBusinessComponents() {
        // 🤖 Wire business components to enterprise infrastructure
        setBusinessValidator(businessValidator);
        setBusinessRules(businessRules);
    }

    // 👨‍💻 REQUIRED: AbstractEnterpriseService Implementation
    @Override
    protected CustomerRepository getRepository() {
        return customerRepository;
    }

    @Override
    protected String getEntityName() {
        return "Customer";
    }

    @Override
    protected Class<Customer> getEntityClass() {
        return Customer.class;
    }

    @Override
    protected Customer createEntityFromRequest(CustomerRequest createDTO) {
        return customerMapper.toEntity(createDTO);
    }

    @Override
    protected void updateEntityFromRequest(CustomerRequest updateDTO, Customer entity) {
        customerMapper.updateEntityFromRequest(updateDTO, entity);
    }

    @Override
    protected CustomerResponse convertToResponse(Customer entity) {
        return customerMapper.toResponse(entity);
    }

    @Override
    protected Page<CustomerResponse> convertToResponsePage(Page<Customer> entityPage) {
        return entityPage.map(customerMapper::toResponse);
    }

    @Override
    protected List<CustomerResponse> convertToResponseList(List<Customer> entities) {
        return customerMapper.toResponseList(entities);
    }

    @Override

        return entity.getId();
    }

    // 👨‍💻 BUSINESS SERVICE INTERFACE IMPLEMENTATION
    @Override
    public CustomerResponse createCustomer(CustomerRequest request) {
        return create(request); // Delegates to AbstractEnterpriseService
    }

    @Override
    public Optional<CustomerResponse> getCustomerById(String id) {
        return findById(id); // Delegates to AbstractEnterpriseService
    }

    @Override
    public Page<CustomerResponse> getAllCustomers(Pageable pageable) {
        return findAll(pageable); // Delegates to AbstractEnterpriseService
    }

    @Override
    public Optional<CustomerResponse> updateCustomer(String id, CustomerRequest request) {
        return update(id, request); // Delegates to AbstractEnterpriseService
    }

    @Override
    public boolean deleteCustomer(String id) {
        return delete(id); // Delegates to AbstractEnterpriseService
    }

    // 👨‍💻 BUSINESS-SPECIFIC METHODS (Add your custom business methods here)
```

```java
@RestController
@RequestMapping("/api/customers")
@Tag(name = "Customer", description = "Customer management operations")
public class CustomerController
    extends AbstractEnterpriseController<Customer, String, CustomerRequest, CustomerRequest, CustomerResponse> {

    private final CustomerService customerService;

    @Autowired
    public CustomerController(CustomerService customerService) {
        this.customerService = customerService;
    }

    // 👨‍💻 REQUIRED: AbstractEnterpriseController Implementation
    @Override
    protected AbstractEnterpriseService<Customer, String, CustomerRequest, CustomerRequest, CustomerResponse> getEnterpriseService() {
        return (AbstractEnterpriseService<Customer, String, CustomerRequest, CustomerRequest, CustomerResponse>) customerService;
    }

    @Override
    protected String getEntityName() {
        return "Customer";
    }

    @Override
    protected String getEntityPathName() {
        return "customers";
    }

    @Override
    protected String extractEntityId(CustomerResponse responseDTO) {
        return responseDTO.getId();
    }

    @Override
    protected List<CustomerResponse> performSearch(Map<String, String> searchParams) {
        String searchText = searchParams.get("search");
        if (searchText != null && !searchText.trim().isEmpty()) {
            return customerService.searchCustomers(searchText);
        }
        return List.of();
    }

    // 👨‍💻 BUSINESS-SPECIFIC ENDPOINTS
    @GetMapping("/vip")
    @Operation(summary = "Get VIP customers", description = "Retrieve all VIP customers")
    @ApiResponse(responseCode = "200", description = "VIP customers retrieved successfully")
    public ResponseEntity<List<CustomerResponse>> getVIPCustomers() {
        return ResponseEntity.ok(customerService.getVIPCustomers());
    }

    @GetMapping("/status/{status}")
    @Operation(summary = "Get customers by status", description = "Retrieve customers with specific status")
    @ApiResponse(responseCode = "200", description = "Customers retrieved successfully")
    public ResponseEntity<List<CustomerResponse>> getCustomersByStatus(
            @Parameter(description = "Customer status") @PathVariable String status) {
        return ResponseEntity.ok(customerService.getCustomersByStatus(status));
    }

    @PostMapping("/{id}/activate")
    @Operation(summary = "Activate customer", description = "Activate a customer account")
    @ApiResponses(value = {
            @ApiResponse(responseCode = "200", description = "Customer activated successfully"),
            @ApiResponse(responseCode = "404", description = "Customer not found")
    })
    public ResponseEntity<CustomerResponse> activateCustomer(@PathVariable String id) {
        return customerService.activateCustomer(id)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }
```
