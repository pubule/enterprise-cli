    @Column(nullable = false)
    private Integer stockQuantity;

    @Column
    private Integer minimumStock = 5;

    @ElementCollection
    @CollectionTable(name = "product_images")
    private List<String> imageUrls = new ArrayList<>();

    @Enumerated(EnumType.STRING)
    private ProductStatus status = ProductStatus.ACTIVE;

    // 👨‍💻 Business relationships
    @ManyToOne(optional = false)
    @JoinColumn(name = "category_id")
    private Category category;

    @ManyToMany
    @JoinTable(name = "product_authors")
    private Set<Author> authors = new HashSet<>();

    // 👨‍💻 Add business methods
    public boolean isInStock() {
        return stockQuantity > 0;
    }

    public boolean isLowStock() {
        return stockQuantity <= minimumStock;
    }

    public void reduceStock(int quantity) {
        if (stockQuantity < quantity) {
            throw new InsufficientStockException("Not enough stock available");
        }
        this.stockQuantity -= quantity;
    }
}

@Service
@Transactional
public class ProductService extends BaseService<Product, Long> {

    // 🤖 Generated basic CRUD methods
    // findAll(), findById(), save(), delete() etc.

    // 👨‍💻 Add business-specific methods

    @Transactional(readOnly = true)
    public Page<Product> findProductsByCategory(
            Long categoryId, Pageable pageable) {
        return productRepository.findByCategoryIdAndStatus(
                categoryId, ProductStatus.ACTIVE, pageable);
    }

    @Transactional(readOnly = true)
    public List<Product> findLowStockProducts() {
        return productRepository.findLowStockProducts();
    }

    @Transactional
    public Product updateStock(Long productId, int quantity) {
        Product product = findById(productId);

        // 👨‍💻 Business validation
        if (quantity < 0) {
            throw new InvalidStockQuantityException("Stock quantity cannot be negative");
        }

        product.setStockQuantity(quantity);

        // 👨‍💻 Business events
        if (product.isLowStock()) {
            publishEvent(new LowStockEvent(product));
        }

        return productRepository.save(product);
    }

    @Transactional
    public List<Product> searchProducts(String query, Pageable pageable) {
        // 👨‍💻 Implement business search logic
        return productRepository.searchByTitleOrAuthorOrIsbn(query, pageable);
    }

    @EventListener
    public void handleOrderPlaced(OrderPlacedEvent event) {
        // 👨‍💻 Update stock when order is placed
        for (OrderItem item : event.getOrderItems()) {
            Product product = findById(item.getProductId());
            product.reduceStock(item.getQuantity());
            productRepository.save(product);
        }
    }
}

import * as yup from 'yup';

export const productFormSchema = yup.object({
  // 👨‍💻 Business-specific validation rules

    .required('Title is required')
    .min(2, 'Title must be at least 2 characters')
    .max(200, 'Title cannot exceed 200 characters'),

  isbn: yup
    .string()
    .required('ISBN is required')
    .matches(/^(?:ISBN(?:-13)?:? )?(?=[0-9]{13}$|(?=(?:[0-9]+[- ]){4})[- 0-9]{17}$)97[89][- ]?[0-9]{1,5}[- ]?[0-9]+[- ]?[0-9]+[- ]?[0-9]$/, 'Invalid ISBN format')
    .test('unique-isbn', 'ISBN already exists', async (value) => {
      if (!value) return true;
      const exists = await checkIsbnExists(value);
      return !exists;
    }),

  price: yup
    .number()
    .required('Price is required')
    .positive('Price must be positive')
    .max(9999.99, 'Price cannot exceed $9999.99')
    .test('decimal-places', 'Price can have maximum 2 decimal places', (value) => {
      if (!value) return true;
      return Number.isInteger(value * 100);
    }),

  description: yup
    .string()
    .required('Description is required')
    .min(10, 'Description must be at least 10 characters')
    .max(2000, 'Description cannot exceed 2000 characters'),

  categoryId: yup
    .number()
    .required('Category is required')
    .positive('Please select a valid category'),

  authorIds: yup
    .array()
    .of(yup.number().positive())
    .min(1, 'At least one author is required')
    .max(5, 'Maximum 5 authors allowed'),

  stockQuantity: yup
    .number()
    .required('Stock quantity is required')
    .integer('Stock must be a whole number')
    .min(0, 'Stock cannot be negative'),

  minimumStock: yup
    .number()
    .integer('Minimum stock must be a whole number')
    .min(0, 'Minimum stock cannot be negative')
    .max(yup.ref('stockQuantity'), 'Minimum stock cannot exceed current stock'),

  images: yup
    .array()
    .of(yup.mixed<File>())
    .max(5, 'Maximum 5 images allowed')
    .test('file-size', 'Each image must be less than 5MB', (files) => {
      if (!files) return true;
      return files.every(file => file instanceof File && file.size <= 5 * 1024 * 1024);
    })
    .test('file-type', 'Only JPEG, PNG, and WebP images are allowed', (files) => {
      if (!files) return true;
      const allowedTypes = ['image/jpeg', 'image/png', 'image/webp'];
      return files.every(file => file instanceof File && allowedTypes.includes(file.type));
    }),

// src/services/apiClient.ts
import axios, { AxiosError, AxiosRequestConfig } from 'axios';

// 🤖 Generated base API client with your enhancements
export const apiClient = axios.create({
  baseURL: process.env.REACT_APP_API_BASE_URL,
  timeout: 10000,
});

// 👨‍💻 Add retry logic for specific errors
const retryableErrors = [
  'NETWORK_ERROR',
  'TIMEOUT',
  'SERVER_ERROR_500',
  'BAD_GATEWAY_502',
  'SERVICE_UNAVAILABLE_503',
  'GATEWAY_TIMEOUT_504'
];

const shouldRetry = (error: AxiosError): boolean => {
  if (!error.response) return true; // Network error

  const status = error.response.status;
  return status >= 500 || retryableErrors.some(code =>
    error.code?.includes(code) || error.message.includes(code)
  );
};

const delay = (ms: number) => new Promise(resolve => setTimeout(resolve, ms));

const retryRequest = async (
  originalRequest: AxiosRequestConfig,
  retryCount = 0,
  maxRetries = 3
): Promise<any> => {
  try {
    return await apiClient.request(originalRequest);
  } catch (error) {
    if (retryCount < maxRetries && shouldRetry(error as AxiosError)) {
      const delayMs = Math.pow(2, retryCount) * 1000; // Exponential backoff
      await delay(delayMs);
      return retryRequest(originalRequest, retryCount + 1, maxRetries);
    }
    throw error;
  }
};

// 👨‍💻 Response interceptor for error handling
apiClient.interceptors.response.use(
  (response) => response,
  async (error: AxiosError) => {
    // Log error for monitoring
    errorTrackingService.logAPIError({
      url: error.config?.url,
      method: error.config?.method,
      status: error.response?.status,
      message: error.message,
      timestamp: new Date().toISOString(),
    });

    // Handle specific business errors
    if (error.response?.status === 401) {
      // Token expired - attempt refresh
      try {
        await refreshAuthToken();
        return retryRequest(error.config!);
      } catch (refreshError) {
        // Redirect to login
        window.location.href = '/login';
        return Promise.reject(refreshError);
      }
    }

    // Handle rate limiting
    if (error.response?.status === 429) {
      const retryAfter = error.response.headers['retry-after'];
      if (retryAfter) {
        await delay(parseInt(retryAfter) * 1000);
        return retryRequest(error.config!);
      }
    }

    // Handle temporary server errors with retry
    if (shouldRetry(error)) {
      return retryRequest(error.config!);
    }

    // Transform error for frontend consumption
    const transformedError = transformAPIError(error);
    return Promise.reject(transformedError);
  }
);

// 👨‍💻 Transform API errors to user-friendly messages
const transformAPIError = (error: AxiosError): AppError => {
  const response = error.response;

  if (!response) {
    return new AppError(
      'Network error. Please check your internet connection.',
      'NETWORK_ERROR',
      'network'
    );
  }

  const data = response.data as any;

  // Use backend error messages when available
  if (data?.message) {
    return new AppError(
      data.message,
      data.error || 'UNKNOWN_ERROR',
      data.errorType || 'general',
      data.suggestions
    );
  }

  // Fallback error messages
  switch (response.status) {
    case 400:
      return new AppError(
        'Invalid request. Please check your input.',
        'BAD_REQUEST',
        'validation'
      );
    case 403:
      return new AppError(
        'You don\'t have permission to perform this action.',
        'FORBIDDEN',
        'permission'
      );
    case 404:
      return new AppError(
        'The requested resource was not found.',
        'NOT_FOUND',
        'notFound'
      );
    case 409:
      return new AppError(
        'This action conflicts with the current state.',
        'CONFLICT',
        'business'
      );
    default:
      return new AppError(
        'An unexpected error occurred. Please try again.',
        'UNKNOWN_ERROR',
        'general'
      );
  }
};

export class AppError extends Error {
  constructor(
    message: string,
    public code: string,
    public type: string,
    public suggestions: string[] = []
  ) {
    super(message);
    this.name = 'AppError';
  }
}
